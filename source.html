  if(state.effect==='none'||$('reduceMotion').checked)throw Error('움직임 효과를 선택하고 움직임 줄이기를 꺼 주세요.');
  if(recording)throw Error('영상을 이미 만들고 있습니다.');

  const stream=c.captureStream(30),recorder=new MediaRecorder(stream,{mimeType:videoMime(),videoBitsPerSecond:6000000}),chunks=[];let frameId=0,abortReason=null;
  recording=true;syncStudio();updateSelection();document.querySelectorAll('.controls input,.controls select,.controls textarea,[data-ratio],[data-preset],[data-effect],#sample,#removeImage,#resetStyle,#createTemplate,#updateTemplate,#exportJSON,#jsonFile').forEach(el=>el.disabled=true);

  document.addEventListener('visibilitychange',visibility);
  try{return await new Promise((resolve,reject)=>{
    recorder.ondataavailable=e=>{if(e.data.size)chunks.push(e.data)};
    recorder.onerror=()=>{abortReason=Error('브라우저가 영상을 만들지 못했습니다. PNG로 저장하거나 다시 시도해 주세요.');if(recorder.state!=='inactive')recorder.stop();else reject(abortReason)};
    recorder.onstop=()=>{cancelAnimationFrame(frameId);const blob=new Blob(chunks,{type:'video/webm'});if(abortReason)reject(abortReason);else if(!blob.size)reject(Error('영상이 비어 있습니다. 다시 시도해 주세요.'));else resolve(blob)};
    recorder.start(100);const start=performance.now();const tick=now=>{try{const p=Math.min(1,(now-start)/3000);draw(card,source,c,{progress:p});draw(card,source,$('preview'),{progress:p});if(p<1)frameId=requestAnimationFrame(tick);else if(recorder.state!=='inactive')recorder.stop()}catch(e){abortReason=e;if(recorder.state!=='inactive')recorder.stop()}};frameId=requestAnimationFrame(tick);
  })}finally{
    cancelAnimationFrame(frameId);stream.getTracks().forEach(t=>t.stop());document.removeEventListener('visibilitychange',visibility);recording=false;
    document.querySelectorAll('.controls input,.controls select,.controls textarea,[data-ratio],[data-preset],[data-effect],#sample,#removeImage,#resetStyle,#createTemplate,#updateTemplate,#exportJSON,#jsonFile').forEach(el=>el.disabled=false);
    if(!db){$('createTemplate').disabled=true;$('jsonFile').disabled=true}$('updateTemplate').disabled=!selected;render();syncStorageControls();
  }
}

# FRAME LAB — 짤·카드 스튜디오

공개 편집기: https://h0p2.github.io/frame-lab-card-studio/

GitHub 소스: https://github.com/H0P2/frame-lab-card-studio

HTML 소스 보기: https://h0p2.github.io/frame-lab-card-studio/source.html

과제 3 결과물. HTML/CSS/JavaScript, Canvas, IndexedDB로 구현했으며 외부 라이브러리를 사용하지 않습니다.

## 실행

이 디렉터리를 정적 HTTP 서버로 제공한 뒤 index.html을 여세요. 파일 더블클릭 실행은 모듈 또는 브라우저 저장소 제한이 있을 수 있습니다.

## 기능과 한도

- PNG/JPEG 실제 파일 서명 검증. 최대 8MB·2천만 픽셀. 긴 변을 최대 2400px로 정리합니다.
- Canvas 재인코딩으로 원본의 EXIF·위치 메타데이터를 제거합니다.
- 문구 최대 2,000자. 한글·영문·줄바꿈·복합 이모지를 처리합니다.
- 요청 크기 20~160px. 넘치면 자동 줄바꿈과 최소 12px까지 축소합니다. 수용 불가 입력은 거부하고 기존 작업을 유지합니다.
- 1:1(1080×1080), 4:5(1080×1350), 9:16(1080×1920). 미리보기 Canvas를 PNG로 직접 저장합니다.
- 사용자 템플릿 생성·불러오기·수정·삭제·최근 삭제 취소. 이미지와 설정을 IndexedDB에 보관합니다.
- 최대 24개, 전체 JSON 32MB 이내. 정상 JSON은 안정 ID로 병합합니다. 전체 검증 실패 시 저장하지 않습니다.
- 작성 중인 카드는 변경 후 400ms 동안 입력이 멈추면 IndexedDB에 자동 저장되고 새로고침 뒤 복구됩니다. 저장 공간이 부족하면 화면에 알립니다. 템플릿도 유지됩니다. 기기를 바꾸면 JSON으로 옮기세요.
- 문구를 직접 드래그하거나 방향키로 이동합니다. Shift 방향키는 20px, 기본은 4px입니다. 카드 안전 영역으로 제한됩니다.
- 실행 취소·다시 실행 최대 40단계, 연속 입력은 묶음 처리합니다. 새 카드도 실행 취소할 수 있습니다.
- 네온·필름·미니멀·매거진 프리셋. 이미지·문구를 보존하고 필터와 배치를 적용합니다.
- 페이드·슬라이드·타이핑·줌 미리보기와 약 3초 WebM 영상 저장. PNG는 움직임이 끝난 완성 장면을 저장합니다. MP4/GIF 저장은 제공하지 않습니다.
- 움직임 줄이기는 즉시 재생을 멈추고 영상 저장을 비활성화합니다. 운영체제 설정도 반영합니다.
- 영상 생성 중 편집을 잠그고, 다른 화면으로 이동하면 생성 취소를 알립니다. WebM 미지원 브라우저는 PNG와 미리보기를 사용합니다.
- 기존 템플릿 JSON은 새 항목 없이도 불러옵니다. 템플릿 저장소는 기존 버전 그대로 연결하며 자동 복구는 별도 저장소를 사용합니다. 이전 버전의 자동 복구 데이터도 이관합니다.
- 편집기는 불러온 이미지 파일을 별도 서버로 업로드하지 않습니다. 브라우저 데이터 삭제 또는 시크릿 창 종료 시 로컬 기록이 사라질 수 있습니다.

## 실제 검사

evidence 폴더에 실제 브라우저 실행 결과와 내려받은 파일을 보관했습니다. 개인정보 없는 합성 입력을 사용했습니다.

수정 전 긴 한글 한 줄 폭 41886.09375px > 사용 폭 936px로 FAIL. 동일 문구에 자동 줄바꿈·크기 보정을 적용하여 33px·20줄로 PASS.

세 화면비에서 미리보기와 PNG 디코딩 결과의 전체 픽셀 채널을 대조했습니다. 템플릿 CRUD·새로고침 유지·잘못된 JSON 거부도 실제 IndexedDB에서 실행했습니다.

## 제작·권리

본 프로젝트 내부에서 AI의 코드 구현 지원으로 제작한 추상 Canvas 그래픽과 문구입니다. 외부 사진·캐릭터·상표를 사용하지 않았습니다. 학생이 직접 그렸다고 주장하지 않습니다. 제작물은 MIT로 제공합니다.

## Studio 추가 검사

evidence/studio에 새 기능 18건과 기존 극단 입력·비율·템플릿·화면·편집 재검사 원본을 보관합니다. WebM 4개는 실제 MediaRecorder로 생성한 뒤 브라우저에서 디코딩했습니다. 드래그 검사는 실제 편집기의 PointerEvent 처리기를 실행했습니다. 키보드·프리셋 UI는 별도 브라우저 조작으로도 확인합니다. 색상 검사는 문구 영역의 실제 RGB 픽셀을 확인합니다.


## 화면 디자인 정리 (2026-10-05)

CSS로 작업 공간과 패널, 입력창, 버튼 위계를 정리했습니다. 이 CSS 정리 단계 당시 동작 HTML·JavaScript는 Studio 버전 2와 동일했습니다. 이후 아래 클릭 복구 단계에서 초기화와 저장소 연결 코드를 수정했습니다. 실제 화면 배치 확인과 캡처는 evidence/studio/polish-layout.json 및 polish-screen 파일에 있습니다. 기존 기능 검사 52건을 다시 실행한 것으로 표현하지 않습니다. 공개 결과물·소스 두 주소의 무로그인 확인은 사용자가 보고했습니다.

## 클릭·초기화 복구

이전 버전 탭이 IndexedDB 연결을 유지하면 저장소 버전 업그레이드가 대기하면서 핵심 이벤트 등록까지 멈추던 실제 결함을 재현했습니다. 편집 이벤트를 먼저 등록하고 저장소 연결을 별도로 실행합니다. 템플릿은 기존 저장소를 유지하고 자동 복구는 별도 저장소를 사용합니다. 연결 지연 시 편집·PNG·영상은 계속 사용할 수 있고 연결 복구 버튼을 제공합니다. 연결 도중 입력한 문구를 늦게 복구된 카드가 덮어쓰지 않도록 검사합니다.

빠른 버튼 클릭을 버리지 않고 순서대로 처리합니다. 자동 복구 시 PNG/JPEG 형식과 크기를 검사하고 위치 정보를 담을 수 있는 메타데이터를 제거하면서 압축된 이미지 픽셀을 다시 압축하지 않습니다. JSON에서 가져온 이미지는 별도로 안전하게 재인코딩합니다.

실제 클릭 검사 39건 및 별도의 마우스·Enter·Space·Escape 검사 기록은 evidence/interaction에 있습니다. 검사 입력은 개인정보 없는 합성 자료입니다. 자동 검사 입력의 방식과 다운로드 관찰 방식은 원본 기록에 명시했습니다.

최종 수정본에서 기존 편집·출력·템플릿·화면·Studio 기능 52건을 다시 실행하여 통과했습니다. 원본은 evidence/interaction/regression-all.json에 있습니다. 자동 복구 검사에서는 설정과 최종 PNG 바이트가 동일했습니다.
